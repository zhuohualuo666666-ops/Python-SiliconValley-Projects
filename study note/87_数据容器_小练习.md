fruits = {
    '苹果': 4.5,
    '香蕉': 3.2,
    '橙子': 5.8,
    '草莓': 12.0,
    '哈密瓜': 8.8
}
key = max(fruits, key=fruits.get)
print(f'最贵的水果是{key}，价格是{fruits[key]} 元/斤')



# 括号里key处理的是待比较元素，fruits.get获取的是字典的value，将其赋给括号里的key是告诉max比较的值是各个字典的value，
# 括号里的fruits是待比较的数据，决定了 max 遍历和比较的原始数据范围
